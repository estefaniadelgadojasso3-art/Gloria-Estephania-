Tarea 5.
Dado el archivo fasta hacer un programa con las siguientes funciones.

def complemento(secuencia):
    pares = {"A": "T", "T": "A", "G": "C", "C": "G",
             "R": "Y", "Y": "R", "K": "M", "M": "K",
             "S": "S", "W": "W", "N": "N"}
    resultado = ""
    for letra in secuencia:
        if letra in pares:
            resultado = resultado + pares[letra]
        else:
            resultado = resultado + letra
    return resultado

def contar_gc(secuencia):
    # Cuenta cuántas G y C hay
    total = 0
    for letra in secuencia:
        if letra == "G" or letra == "C":
            total = total + 1
    return total

def formato_gc(valor):
    # Corta a 2 decimales: 0.0909 -> .09, 0 -> 0, 1 -> 1
    n = int(valor * 100)
    if n == 0:
        return "0"
    if n == 100:
        return "1"
    if n < 10:
        return ".0" + str(n)
    return "." + str(n)

nombre_archivo = "AJ278755.fasta"
organismo = nombre_archivo.replace(".fasta", "")

archivo = open(nombre_archivo, "r")

numero_linea = 0
primera = False      # True cuando la siguiente secuencia es la primera del segmento
gc_total = 0
bases_total = 0

for linea in archivo:
    numero_linea = numero_linea + 1
    linea = linea.strip()

    if linea == "":
        continue

    if linea.startswith(">"):
        # 1. Encabezado
        print("Organismo " + organismo + ", contig " + linea[1:])
        primera = True
    else:
        linea = linea.upper()

        # 2. GC de la línea
        gc = contar_gc(linea)
        print("GC línea " + str(numero_linea) + ": " + formato_gc(gc / len(linea)))

        # Acumular para el total
        gc_total = gc_total + gc
        bases_total = bases_total + len(linea)

        # 3. Complementaria de la primera secuencia del segmento
        if primera:
            print(complemento(linea))
            primera = False

        # 4. Buscar ATG
        if "ATG" in linea:
            print("ATG en línea " + str(numero_linea))

archivo.close()

# 5. GC total
print("GC total del archivo: " + formato_gc(gc_total / bases_total))
